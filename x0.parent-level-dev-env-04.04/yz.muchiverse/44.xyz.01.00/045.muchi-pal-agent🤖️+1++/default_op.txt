out void builtin_out {print register to stdout}
halt void builtin_halt {stop execution}
gemma_strategy void ops/+x/gemma_strategy.+x {pre-Gemma deterministic tool detection - runs BEFORE send_message on every Enter, see gemma_strategy.c's own header comment}
strategy_execute_a void ops/+x/strategy_execute_a.+x {pre-execute the tool gemma_strategy.+x just detected (Strategy A), before any LLM call}
send_message void ops/+x/send_message.+x {Enter handler - send a chat message, run a /model command, or answer a pending y/n}
check_response void ops/+x/check_response.+x {poll for a completed API response and parse it}
execute_tool void ops/+x/execute_tool.+x {run the pending tool call after user approval}
deny_tool void ops/+x/deny_tool.+x {record a denied tool call}
switch_model void ops/+x/switch_model.+x {look up a model id in the registry and switch to it}
compose_frame void ops/+x/compose_frame.+x {render the conversation tail + status into view.txt}
set_irc_agent void ops/+x/set_irc_agent.+x {"/irc-agent <room>|off" - turn the pal-chat-irc bridge on/off}
irc_agent_poll void ops/+x/irc_agent_poll.+x {tick-driven no-op unless irc_agent_active=yes - joins/greets/auto-responds in pal-chat-irc}
